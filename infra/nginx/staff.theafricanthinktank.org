# Map CORS pour les deux domaines
map $http_origin $cors_origin {
    default "";
    "https://staff.theafricanthinktank.org" "$http_origin";
    "https://staff.theafricanthinktank.com" "$http_origin";
}

# Redirection HTTP vers HTTPS pour les deux domaines
server {
    if ($host = staff.theafricanthinktank.com) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    if ($host = staff.theafricanthinktank.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    server_name staff.theafricanthinktank.org staff.theafricanthinktank.com;
    return 301 https://$server_name$request_uri;




}

# Serveur HTTPS
server {
    server_name staff.theafricanthinktank.org staff.theafricanthinktank.com;

    location /adminer/ {
        proxy_pass http://localhost:8080/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location / {
        proxy_pass http://localhost:3001;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location /api {
        add_header 'Access-Control-Allow-Origin' $cors_origin always;
        add_header 'Access-Control-Allow-Methods' 'GET, POST, PUT, PATCH, DELETE, OPTIONS' always;
        add_header 'Access-Control-Allow-Headers' 'Content-Type, Authorization, Accept' always;
        add_header 'Access-Control-Allow-Credentials' 'true' always;

        if ($request_method = 'OPTIONS') {
            add_header 'Access-Control-Allow-Origin' $cors_origin always;
            add_header 'Access-Control-Allow-Methods' 'GET, POST, PUT, PATCH, DELETE, OPTIONS' always;
            add_header 'Access-Control-Allow-Headers' 'Content-Type, Authorization, Accept' always;
            add_header 'Access-Control-Allow-Credentials' 'true' always;
            add_header 'Content-Length' 0;
            return 204;
        }

        rewrite ^/api/(.*) /$1 break;
        proxy_pass http://localhost:5000/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    listen 443 ssl;
    ssl_certificate /etc/letsencrypt/live/staff.theafricanthinktank.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/staff.theafricanthinktank.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;


}
